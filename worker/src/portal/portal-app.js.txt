/* Provider portal client — renders /portal/data into the ledger UI.
 * Whole-school codes get a cohort switcher (client-side re-filter);
 * tag-scoped codes are filtered server-side and locked to their tag. */
(function () {
  "use strict";
  var $ = function (id) { return document.getElementById(id); };
  var TIER = {
    high: { l: "Needs contact", c: "#D9452B" },
    medium: { l: "Drifting", c: "#ED9249" },
    watch: { l: "Early wobble", c: "#13507F" },
    new: { l: "New", c: "#8a97a1" },
    ok: { l: "Engaged", c: "#1B7A4B" },
  };
  var ORDER = ["high", "medium", "watch", "new", "ok"];
  function esc(t) {
    return String(t).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;").replace(/'/g, "&#39;");
  }

  /* ---- tabs ---- */
  var tabs = document.querySelectorAll(".ltab");
  function goTab(name) {
    tabs.forEach(function (x) {
      var on = x.dataset.v === name;
      x.classList.toggle("on", on);
      x.setAttribute("aria-selected", String(on));
    });
    document.querySelectorAll(".lview").forEach(function (v) { v.classList.remove("on"); });
    $("lv-" + name).classList.add("on");
    window.scrollTo({ top: 0 });
  }
  tabs.forEach(function (t) {
    t.addEventListener("click", function () { goTab(t.dataset.v); });
  });
  var aAll = $("action-all");
  if (aAll) aAll.addEventListener("click", function () { goTab("warning"); });
  var mAll = $("mods-all");
  if (mAll) mAll.addEventListener("click", function () { goTab("modules"); });

  var DATA = null;
  var FILTER = null; // client-side cohort filter (whole-school codes only)

  function learners() {
    var all = (DATA.risk.learners || []);
    if (!FILTER) return all;
    if (FILTER === "__none__") return all.filter(function (a) { return !a.tags || a.tags.length === 0; });
    return all.filter(function (a) { return (a.tags || []).indexOf(FILTER) !== -1; });
  }

  function summarise(list) {
    var s = { learners: list.length, activeLast7Days: 0, neverLoggedIn: 0, tiers: { high: 0, medium: 0, watch: 0, new: 0, ok: 0 } };
    list.forEach(function (a) {
      s.tiers[a.tier] += 1;
      if (a.daysSinceLogin !== null && a.daysSinceLogin <= 7) s.activeLast7Days += 1;
      if (a.daysSinceLogin === null) s.neverLoggedIn += 1;
    });
    return s;
  }

  function median(list) {
    var days = list
      .map(function (a) { return a.daysSinceLogin; })
      .filter(function (d) { return d !== null; })
      .sort(function (a, b) { return a - b; });
    if (!days.length) return null;
    return days[Math.floor(days.length / 2)];
  }

  /* ---- masthead ledger ---- */
  function renderLedger() {
    var list = learners();
    var sm = FILTER ? summarise(list) : DATA.risk.summary;
    var attn = sm.tiers.high + sm.tiers.medium;
    var activation = sm.learners ? Math.round((sm.learners - sm.neverLoggedIn) * 100 / sm.learners) : 0;
    var med = median(list);
    $("lg-total").textContent = sm.learners;
    $("lg-active").textContent = sm.activeLast7Days;
    var activePct = sm.learners ? Math.round(sm.activeLast7Days * 100 / sm.learners) : 0;
    $("lg-activepct").textContent = sm.learners ? activePct + "%" : "";
    var abar = $("lg-activebar");
    if (abar) abar.style.width = activePct + "%";
    $("lg-activation").textContent = activation + "%";
    $("lg-median").textContent = med === null ? "—" : med + "d";
    $("lg-attn").textContent = attn;
    var h = DATA.history || [];
    var delta = "";
    if (!FILTER && h.length >= 2) {
      var d = h[h.length - 1].activeLast7Days - h[h.length - 2].activeLast7Days;
      if (d !== 0) delta = (d > 0 ? "▲ " : "▼ ") + Math.abs(d) + " vs last snapshot";
    }
    $("lg-delta").textContent = delta;
    var badge = $("t-warnn");
    badge.textContent = attn;
    badge.hidden = attn === 0;
  }

  /* ---- alert strip (safeguarding first, then early warning) ---- */
  function renderAlerts(flagCount) {
    var host = $("alerts");
    if (!host) return;
    var sm = FILTER ? summarise(learners()) : DATA.risk.summary;
    var attn = sm.tiers.high + sm.tiers.medium;
    var out = "";
    if (flagCount > 0) {
      out +=
        "<div class='alert red' data-go='reflect'><span class='ai'>⚠</span><div>" +
        "<div class='at'>Safeguarding flags need review</div>" +
        "<div class='ad'>" + flagCount + " reflection answer" + (flagCount === 1 ? "" : "s") +
        " flagged for crisis language — read " + (flagCount === 1 ? "it" : "them") + " today and follow your policy.</div></div>" +
        "<span class='al'>Review now →</span></div>";
    }
    if (attn > 0) {
      out +=
        "<div class='alert amber' data-go='warning'><span class='ai'>◔</span><div>" +
        "<div class='at'>" + attn + " learner" + (attn === 1 ? "" : "s") + " going quiet</div>" +
        "<div class='ad'>Each has a ready-to-send nudge — a two-minute message now beats a withdrawal call later.</div></div>" +
        "<span class='al'>See who →</span></div>";
    }
    host.innerHTML = out;
    host.querySelectorAll(".alert").forEach(function (a) {
      a.addEventListener("click", function () { goTab(a.dataset.go); });
    });
  }

  /* ---- curriculum impact (framework bars) ---- */
  function renderCurriculum() {
    var host = $("curric");
    if (!host) return;
    var rows = DATA.curriculum || [];
    if (!rows.length) {
      host.innerHTML = "<p class='muted'>Curriculum figures appear once the sample has module activity.</p>";
      return;
    }
    var out = "";
    rows.forEach(function (r) {
      var startedPct = r.enrolled ? 100 : 0;
      out +=
        "<div class='cirow'><span class='cn'>" + esc(r.area) +
        "<em>" + r.completed + " completions · " + r.enrolled + " enrolments</em></span>" +
        "<span class='citrack'><i class='prog' style='width:" + startedPct + "%'></i>" +
        "<i class='done' style='width:" + r.pct + "%'></i></span>" +
        "<span class='cp'>" + r.pct + "%</span></div>";
    });
    host.innerHTML = out;
  }

  /* ---- who needs you today (top 5 most urgent) ---- */
  function renderActions() {
    var list = learners().filter(function (a) {
      return a.tier === "high" || a.tier === "medium" || a.tier === "watch";
    });
    var count = $("action-count");
    count.hidden = list.length === 0;
    count.textContent = list.length + " flagged";
    var top = list.slice(0, 5);
    if (!top.length) {
      $("action-list").innerHTML =
        "<div class='aok'>✓ Nobody needs chasing today — everyone is engaged or brand new.</div>";
      return;
    }
    var out = "";
    top.forEach(function (a, i) {
      var days = a.daysSinceLogin === null ? "never" : a.daysSinceLogin + "d";
      var initials = (a.name || "?").split(/\s+/).map(function (w) { return w.charAt(0); }).slice(0, 2).join("").toUpperCase() || "?";
      out +=
        "<div class='arow'>" +
        "<span class='wava" + (a.tier === "watch" ? " cool" : "") + "'>" + esc(initials) + "</span>" +
        "<span class='aname'>" + esc(a.name) + "<em>" + (a.cohort ? esc(a.cohort) : esc(a.email)) + "</em></span>" +
        "<span class='areason'>" + esc(a.reasons[0] || "") + "</span>" +
        "<span class='wdays " + (a.daysSinceLogin === null || a.daysSinceLogin > 20 ? "hot" : "") + "'>" + days + "</span>" +
        "<button class='abtn2' data-a='" + i + "'>Copy nudge</button>" +
        "</div>";
    });
    $("action-list").innerHTML = out;
    document.querySelectorAll("#action-list .abtn2").forEach(function (b) {
      b.addEventListener("click", function () {
        var a = top[+b.dataset.a];
        function done() { b.textContent = "Copied ✓"; setTimeout(function () { b.textContent = "Copy nudge"; }, 2200); }
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(a.nudge).then(done, function () { window.prompt("Copy the nudge:", a.nudge); });
        } else { window.prompt("Copy the nudge:", a.nudge); }
      });
    });
  }

  /* ---- cohort pulse (stacked bar + plain-English legend rows) ---- */
  var TIER_DEF = {
    high: "21+ days quiet, or never logged in after two weeks",
    medium: "10–20 days since last visit",
    watch: "8–9 days quiet, or active without finishing",
    new: "joined within the last week",
    ok: "active in the last 7 days",
  };
  function renderPulse() {
    var sm = FILTER ? summarise(learners()) : DATA.risk.summary;
    var total = sm.learners || 1;
    var seg = "";
    var rows = "";
    ORDER.forEach(function (k) {
      var n = sm.tiers[k] || 0;
      if (n) seg += "<i style='width:" + (n * 100 / total) + "%;background:" + TIER[k].c + "'></i>";
      rows +=
        "<div class='plr'><i style='background:" + TIER[k].c + "'></i>" +
        "<span class='n'>" + TIER[k].l + "</span>" +
        "<span class='d'>" + TIER_DEF[k] + "</span>" +
        "<span class='v'>" + n + "</span></div>";
    });
    $("pulse-bar").innerHTML = seg;
    $("pulse-legend").innerHTML = rows;
  }

  /* ---- engagement trend (area chart) ---- */
  function renderTrend() {
    var h = DATA.history || [];
    var host = $("trend-svg");
    if (FILTER || DATA.scopedTag || h.length < 2) {
      host.innerHTML = "";
      $("trend-note").textContent = FILTER
        ? "The trend is measured school-wide — clear the cohort filter to see it."
        : DATA.scopedTag
          ? "Your cohort's trend line starts growing from today's snapshot."
          : "The trend appears after a few nightly snapshots — the first is already banked.";
      return;
    }
    var W = 560, H = 150, PAD = 26;
    var pts = h.map(function (p, i) {
      var x = PAD + i * ((W - PAD * 2) / (h.length - 1));
      var pct = p.learners ? p.activeLast7Days / p.learners : 0;
      var y = H - 22 - pct * (H - 48);
      return { x: x, y: y, pct: Math.round(pct * 100), date: p.date };
    });
    var line = pts.map(function (p, i) { return (i ? "L" : "M") + p.x.toFixed(1) + " " + p.y.toFixed(1); }).join(" ");
    var area = line + " L" + pts[pts.length - 1].x.toFixed(1) + " " + (H - 22) + " L" + pts[0].x.toFixed(1) + " " + (H - 22) + " Z";
    var last = pts[pts.length - 1];
    host.innerHTML =
      "<defs><linearGradient id='tg' x1='0' y1='0' x2='0' y2='1'>" +
      "<stop offset='0' stop-color='#D9452B' stop-opacity='.28'/>" +
      "<stop offset='1' stop-color='#D9452B' stop-opacity='0'/></linearGradient></defs>" +
      "<line x1='" + PAD + "' y1='" + (H - 22) + "' x2='" + (W - PAD) + "' y2='" + (H - 22) + "' stroke='#d8d2cf' stroke-width='1'/>" +
      "<path d='" + area + "' fill='url(#tg)'/>" +
      "<path d='" + line + "' fill='none' stroke='#D9452B' stroke-width='2.5' stroke-linejoin='round' stroke-linecap='round'/>" +
      "<circle cx='" + last.x + "' cy='" + last.y + "' r='4.5' fill='#D9452B'/>" +
      "<text x='" + Math.min(last.x, W - 60) + "' y='" + (last.y - 10) + "' font-size='12' font-weight='700' fill='#05253C'>" + last.pct + "% active</text>" +
      "<text x='" + PAD + "' y='" + (H - 6) + "' font-size='10' fill='#8a97a1'>" + pts[0].date.slice(5) + "</text>" +
      "<text x='" + (W - PAD) + "' y='" + (H - 6) + "' font-size='10' fill='#8a97a1' text-anchor='end'>" + last.date.slice(5) + "</text>";
    $("trend-note").textContent = "Share of learners active in the last 7 days, across " + h.length + " nightly snapshots.";
  }

  /* ---- recency ladder ---- */
  function renderRecency() {
    var list = learners();
    var buckets = [
      { l: "Today", f: function (d) { return d === 0; } },
      { l: "This week", f: function (d) { return d !== null && d >= 1 && d <= 7; } },
      { l: "1–2 weeks", f: function (d) { return d !== null && d >= 8 && d <= 14; } },
      { l: "2–4 weeks", f: function (d) { return d !== null && d >= 15 && d <= 30; } },
      { l: "Over a month", f: function (d) { return d !== null && d > 30; } },
      { l: "Never logged in", f: function (d) { return d === null; } },
    ];
    var max = 1;
    var counts = buckets.map(function (b) {
      var n = list.filter(function (a) { return b.f(a.daysSinceLogin); }).length;
      if (n > max) max = n;
      return n;
    });
    var out = "";
    buckets.forEach(function (b, i) {
      var warm = i >= 3;
      out +=
        "<div class='rrow'><span class='rl'>" + b.l + "</span>" +
        "<span class='rb'><i style='width:" + (counts[i] * 100 / max) + "%;background:" + (warm ? "#D9452B" : "#13507F") + "'></i></span>" +
        "<b class='rn'>" + counts[i] + "</b></div>";
    });
    $("recency").innerHTML = out;
  }

  /* ---- most-used modules (whole-school sample) ---- */
  function renderModules() {
    var rows = DATA.stats.courseStats || [];
    var mini = "";
    rows.slice(0, 6).forEach(function (r, i) {
      var pct = r.enrolled ? Math.round(r.completed * 100 / r.enrolled) : 0;
      var pcol = pct >= 70 ? "#1B7A4B" : pct >= 40 ? "#13507F" : "#D9452B";
      mini +=
        "<div class='mmr'><span class='rk'>" + String(i + 1).padStart(2, "0") + "</span>" +
        "<span class='ti'>" + esc(r.title) + "<em>" + r.enrolled + " enrolled</em></span>" +
        "<span class='mbar'><i style='width:" + pct + "%'></i></span>" +
        "<span class='pc' style='color:" + pcol + "'>" + pct + "%</span></div>";
    });
    $("mods-mini").innerHTML = mini || "<p class='muted'>No module activity in the sample yet.</p>";

    var table = "";
    rows.forEach(function (r) {
      var pct = r.enrolled ? Math.round(r.completed * 100 / r.enrolled) : 0;
      table +=
        "<tr><td>" + esc(r.title) + "</td><td class='num'>" + r.enrolled + "</td>" +
        "<td class='num'>" + r.completed + "</td>" +
        "<td class='barcell'><span class='mbar'><i style='width:" + pct + "%'></i></span></td>" +
        "<td class='num'>" + pct + "%</td></tr>";
    });
    $("mods-table").innerHTML =
      "<table><thead><tr><th>Module</th><th class='num'>Enrolled</th><th class='num'>Completed</th><th style='width:30%'>Completion</th><th class='num'>%</th></tr></thead><tbody>" +
      table + "</tbody></table>";
  }

  /* ---- early warning ledger ---- */
  function renderWarning() {
    var list = learners();
    var sm = summarise(list);
    var chips = "<button class='fchip on' data-t='flagged'>Flagged <b>" + (sm.tiers.high + sm.tiers.medium + sm.tiers.watch) + "</b></button>";
    ORDER.forEach(function (k) {
      chips += "<button class='fchip' data-t='" + k + "'><i style='background:" + TIER[k].c + "'></i>" + TIER[k].l + " <b>" + (sm.tiers[k] || 0) + "</b></button>";
    });
    chips += "<button class='fchip' data-t='all'>Everyone <b>" + sm.learners + "</b></button>";
    $("warn-chips").innerHTML = chips;

    var current = "flagged";
    function rowsFor(t) {
      if (t === "all") return list;
      if (t === "flagged") return list.filter(function (a) { return a.tier === "high" || a.tier === "medium" || a.tier === "watch"; });
      return list.filter(function (a) { return a.tier === t; });
    }
    function draw(t) {
      current = t;
      var rows = rowsFor(t);
      if (!rows.length) {
        $("warn-list").innerHTML = "<div class='empty'>Nobody in this view — " + (t === "flagged" ? "everyone is engaged or brand new. Lovely." : "try another filter.") + "</div>";
        return;
      }
      var out = "";
      var lastTier = "";
      rows.forEach(function (a, i) {
        if (a.tier !== lastTier) {
          lastTier = a.tier;
          out += "<div class='wgroup'><i style='background:" + TIER[a.tier].c + "'></i>" + TIER[a.tier].l + "</div>";
        }
        var days = a.daysSinceLogin === null ? "never" : a.daysSinceLogin + "d";
        var initials = (a.name || "?").split(/\s+/).map(function (w) { return w.charAt(0); }).slice(0, 2).join("").toUpperCase() || "?";
        out +=
          "<div class='wrow" + (a.tier === "watch" ? " cool" : "") + "' data-i='" + i + "'>" +
          "<span class='wava'>" + esc(initials) + "</span>" +
          "<span class='wname'>" + esc(a.name) + "<em>" + esc(a.email) + (a.cohort ? " · " + esc(a.cohort) : "") + "</em></span>" +
          "<span class='wreason'>" + esc(a.reasons[0] || "") + "</span>" +
          "<span class='wdays " + (a.daysSinceLogin === null || a.daysSinceLogin > 20 ? "hot" : "") + "'>" + days + "</span>" +
          "<span class='wchev'>›</span></div>" +
          "<div class='wdetail' id='wd-" + i + "'>" +
          (a.reasons.length > 1 ? "<ul>" + a.reasons.slice(1).map(function (r) { return "<li>" + esc(r) + "</li>"; }).join("") + "</ul>" : "") +
          "<blockquote>" + esc(a.nudge) + "</blockquote>" +
          "<button class='copybtn' data-n='" + i + "'>Copy nudge message</button>" +
          "<span class='copied' id='wc-" + i + "' hidden>Copied ✓</span></div>";
      });
      $("warn-list").innerHTML = out;
      document.querySelectorAll(".wrow").forEach(function (r) {
        r.addEventListener("click", function () {
          var d = $("wd-" + r.dataset.i);
          var open = d.classList.contains("on");
          document.querySelectorAll(".wdetail").forEach(function (x) { x.classList.remove("on"); });
          document.querySelectorAll(".wrow").forEach(function (x) { x.classList.remove("open"); });
          if (!open) { d.classList.add("on"); r.classList.add("open"); }
        });
      });
      document.querySelectorAll(".copybtn").forEach(function (b) {
        b.addEventListener("click", function (e) {
          e.stopPropagation();
          var a = rowsFor(current)[+b.dataset.n];
          var ok = $("wc-" + b.dataset.n);
          function done() { ok.hidden = false; setTimeout(function () { ok.hidden = true; }, 2400); }
          if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(a.nudge).then(done, function () { window.prompt("Copy the nudge:", a.nudge); });
          } else { window.prompt("Copy the nudge:", a.nudge); }
        });
      });
    }
    document.querySelectorAll(".fchip").forEach(function (c) {
      c.addEventListener("click", function () {
        document.querySelectorAll(".fchip").forEach(function (x) { x.classList.remove("on"); });
        c.classList.add("on");
        draw(c.dataset.t);
      });
    });
    draw("flagged");
  }

  var FLAG_COUNT = 0;
  function renderAll() {
    renderLedger();
    renderAlerts(FLAG_COUNT);
    renderCurriculum();
    renderActions();
    renderPulse();
    renderTrend();
    renderRecency();
    renderModules();
    renderWarning();
  }

  /* Honest first-load message: a cold cache builds live figures from
   * LearnWorlds and takes a few seconds. */
  var slowNote = setTimeout(function () {
    var m = $("p-meta");
    if (m) m.textContent = "Building your live figures from LearnWorlds — the first view takes a few seconds…";
  }, 1500);
  fetch("/portal/data")
    .then(function (r) { return r.json(); })
    .then(function (d) {
      clearTimeout(slowNote);
      if (d.error) {
        $("p-err").hidden = false;
        $("p-err").textContent = "Could not load data: " + d.error;
        return;
      }
      DATA = d;
      $("p-meta").textContent =
        "Generated " +
        new Date(d.risk.summary.assessedAt).toLocaleString("en-GB", { day: "numeric", month: "short", hour: "2-digit", minute: "2-digit" }) +
        " · refreshes every 6 hours" +
        (d.scopedTag ? " · scoped to " + d.scopedTag : "");
      /* cohort switcher (whole-school codes only) */
      var sel = $("cohort-sel");
      if (!d.scopedTag && d.cohorts && d.cohorts.length) {
        sel.hidden = false;
        var opts = "<option value=''>Whole school</option>";
        d.cohorts.forEach(function (c) {
          opts += "<option value='" + esc(c.tag) + "'>" + esc(c.tag) + " (" + c.count + ")</option>";
        });
        var untagged = (d.risk.learners || []).filter(function (a) { return !a.tags || !a.tags.length; }).length;
        if (untagged) opts += "<option value='__none__'>Untagged (" + untagged + ")</option>";
        sel.innerHTML = opts;
        sel.addEventListener("change", function () {
          FILTER = sel.value || null;
          renderAll();
        });
      }
      /* Narrative loads lazily — it needs a model call and must never
       * hold up the dashboard. */
      $("narrative").textContent = "Fledge is drafting your evidence narrative…";
      var NARR = "";
      fetch("/portal/narrative")
        .then(function (r) { return r.json(); })
        .then(function (n) { NARR = n.narrative || ""; $("narrative").textContent = NARR; })
        .catch(function () { $("narrative").textContent = "Narrative unavailable just now — the figures on the dashboard are live."; });
      $("copy-narr").addEventListener("click", function () {
        function done() { $("copied-narr").hidden = false; setTimeout(function () { $("copied-narr").hidden = true; }, 2400); }
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(NARR).then(done);
        } else { window.prompt("Copy:", NARR); }
      });
      /* Inspector link: signed, 7-day, aggregate-only snapshot URL. */
      var insp = $("inspect-link");
      if (insp) insp.addEventListener("click", function () {
        insp.disabled = true;
        fetch("/portal/inspect-link", { method: "POST" })
          .then(function (r) { return r.json(); })
          .then(function (j) {
            insp.disabled = false;
            if (!j || !j.ok) return;
            var full = location.origin + j.url;
            function done() { $("copied-inspect").hidden = false; setTimeout(function () { $("copied-inspect").hidden = true; }, 2600); }
            if (navigator.clipboard && navigator.clipboard.writeText) {
              navigator.clipboard.writeText(full).then(done, function () { window.prompt("Inspector link:", full); });
            } else { window.prompt("Inspector link:", full); }
          })
          .catch(function () { insp.disabled = false; });
      });
      renderAll();
      document.body.classList.add("loaded");
    })
    .catch(function () {
      $("p-err").hidden = false;
      $("p-err").textContent = "Could not reach the portal service — try refreshing.";
    });

  /* ---------- reflections + safeguarding flags ---------- */
  function fmtWhen(ts) {
    if (!ts) return "";
    try {
      return new Date(ts * 1000).toLocaleDateString("en-GB", { day: "numeric", month: "short" });
    } catch (e) { return ""; }
  }
  function renderCoverage(d) {
    var cov = d.coverage || [];
    if (!cov.length) { $("coverage-sect").hidden = true; return; }
    $("coverage-sect").hidden = false;
    var out = "<div class='cvhead'><span>Module</span><span>Pre reflection</span><span>Post reflection</span></div>";
    cov.forEach(function (cv) {
      var pre = cv.preTitle
        ? "<span class='cvu ok'><i>✓</i>" + esc(cv.preTitle) + "</span>"
        : "<span class='cvu miss'><i>–</i>not found</span>";
      var post = cv.postTitle
        ? "<span class='cvu ok'><i>✓</i>" + esc(cv.postTitle) + "</span>"
        : "<span class='cvu miss'><i>–</i>not found</span>";
      out += "<div class='cvrow'><span class='m'>" + esc(cv.courseTitle) + "</span>" + pre + post + "</div>";
    });
    $("coverage-list").innerHTML = out;
  }
  function renderReflections(d) {
    var state = $("reflect-state"), body = $("reflect-body");
    if (d.error) {
      body.hidden = true;
      state.innerHTML = "<div class='err'>Could not load reflections — try refreshing.</div>";
      return;
    }
    if (d.status === "building") {
      body.hidden = true;
      state.innerHTML =
        "<div class='notice'><b>Reading reflections…</b>Scanned " + d.progress.done + " of " +
        d.progress.total + " modules. This first sweep takes a minute — the tab updates itself.</div>";
      renderCoverage(d);
      setTimeout(loadReflections, 2500);
      return;
    }
    renderCoverage(d);
    if (d.responsesEnabled === false) {
      body.hidden = true;
      state.innerHTML =
        "<div class='notice' style='margin-bottom:26px'><b>Answers need one switch flipping at LearnWorlds</b>" +
        "The reflection units below are all located and ready — but the LearnWorlds plan hasn't got the " +
        "Assessments API endpoints enabled, so their answers can't be read yet. Send LearnWorlds support this:" +
        "<div class='ask' id='lw-ask'>" + esc(d.reason || "") + "</div>" +
        "<button class='copybtn' id='copy-ask'>Copy message</button><span class='copied' id='copied-ask' hidden>Copied ✓</span>" +
        "<div style='margin-top:10px;font-size:12.5px;color:#7d8a93'>The moment it's enabled, this tab fills itself in — no rebuild needed.</div></div>";
      var btn = document.getElementById("copy-ask");
      if (btn) btn.addEventListener("click", function () {
        var t = d.reason || "";
        function done() { document.getElementById("copied-ask").hidden = false; }
        if (navigator.clipboard && navigator.clipboard.writeText) { navigator.clipboard.writeText(t).then(done); }
        else { window.prompt("Copy:", t); }
      });
      return;
    }
    /* ready with answers */
    state.innerHTML = "";
    body.hidden = false;
    var flags = d.flags || [];
    var badge = $("t-flagn");
    if (flags.length) { badge.hidden = false; badge.textContent = flags.length; }
    FLAG_COUNT = flags.length;
    if (DATA) renderAlerts(FLAG_COUNT);
    var sg = "";
    flags.forEach(function (f) {
      sg +=
        "<div class='sgrow'><div class='who'>" + esc(f.email) +
        (f.cohort ? " <span class='wdays'>" + esc(f.cohort) + "</span>" : "") + "</div>" +
        "<div class='whr'>" + esc(f.courseTitle) + " · " + esc(f.unitTitle) +
        (f.submittedAt ? " · " + fmtWhen(f.submittedAt) : "") + "</div>" +
        "<div class='q'>" + esc(f.question || "Free-text answer") + "</div>" +
        "<blockquote>" + esc(f.answer) + "</blockquote></div>";
    });
    $("sg-list").innerHTML = flags.length ? sg :
      "<div class='sgok'>✓ No crisis language detected in any reflection answers" +
      (d.scoped ? " in your cohort" : "") + ".</div>";
    $("reflect-meta").textContent =
      d.preCount + " learner" + (d.preCount === 1 ? "" : "s") + " completed an initial reflection · " +
      d.postCount + " completed post-module feedback" + (d.scoped ? " (your cohort)" : "");
    var shifts = (d.shifts || []).slice().sort(function (a, b) {
      return (b.shift === null ? -999 : b.shift) - (a.shift === null ? -999 : a.shift);
    });
    var out = "";
    shifts.forEach(function (s) {
      var hasBoth = s.preAvgPct !== null && s.postAvgPct !== null;
      var up = hasBoth && s.shift >= 0;
      var lo = hasBoth ? Math.min(s.preAvgPct, s.postAvgPct) : 0;
      var hi = hasBoth ? Math.max(s.preAvgPct, s.postAvgPct) : 0;
      out +=
        "<div class='shrow'><span class='shname'>" + esc(s.courseTitle) +
        "<em>" + s.preCount + " pre · " + s.postCount + " post</em></span>" +

        "<span class='rail2'></span>" +
        (hasBoth
          ? "<span class='fill" + (up ? "" : " down") + "' style='left:" + lo + "%;width:" + Math.max(1, hi - lo) + "%'></span>" +
            "<span class='dot pre' style='left:calc(" + s.preAvgPct + "% - 7px)'></span>" +
            "<span class='dot post" + (up ? "" : " down") + "' style='left:calc(" + s.postAvgPct + "% - 7px)'></span>" +
            "<span class='lab' style='left:" + s.preAvgPct + "%'>" + s.preAvgPct + "</span>" +
            "<span class='lab' style='left:" + s.postAvgPct + "%'>" + s.postAvgPct + "</span>"
          : "") +
        "</span>" +
        (hasBoth
          ? "<span class='spill " + (up ? "up" : "down") + "'>" + (s.shift >= 0 ? "+" : "") + s.shift + " pts</span>"
          : "<span class='spill na'>awaiting post data</span>") +
        "</div>";
    });
    $("shift-list").innerHTML = out ||
      "<div class='muted' style='padding:20px 0'>No reflection submissions yet — figures appear as learners complete their pre and post reflections.</div>";
  }
  /* ---------- live activity feed ---------- */
  function ago(ts) {
    var s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
    if (s < 60) return "just now";
    if (s < 3600) return Math.floor(s / 60) + " min ago";
    if (s < 86400) return Math.floor(s / 3600) + "h ago";
    return Math.floor(s / 86400) + "d ago";
  }
  function renderFeed(d) {
    var dot = $("livedot"), hint = $("live-hint"), list = $("feed-list");
    if (!list) return;
    if (!d.configured) {
      dot.className = "livedot";
      hint.textContent = "not connected yet";
      list.innerHTML =
        "<div class='feempty'>Real-time updates switch on once the LearnWorlds webhook is connected — " +
        "completions and new learners will appear here the moment they happen.</div>";
      return;
    }
    dot.className = "livedot on";
    hint.textContent = d.lastEvent ? "streaming from LearnWorlds" : "connected — waiting for first event";
    var rows = "";
    (d.feed || []).forEach(function (f) {
      var ic = f.kind === "completion" ? "✓" : f.kind === "joined" ? "＋" : "★";
      var what =
        f.kind === "completion"
          ? "<b>" + esc(f.name) + "</b> completed <b>" + esc(f.detail) + "</b>"
          : f.kind === "joined"
            ? "<b>" + esc(f.name) + "</b> joined Fledglings"
            : "New lead: <b>" + esc(f.name) + "</b>";
      rows +=
        "<div class='ferow " + f.kind + "'><span class='fico'>" + ic + "</span>" +
        "<div><div class='fetxt'>" + what + (f.cohort ? " · " + esc(f.cohort) : "") + "</div>" +
        "<div class='fetime'>" + ago(f.at) + "</div></div></div>";
    });
    list.innerHTML = rows ||
      "<div class='feempty'>Nothing yet today — activity appears here in real time.</div>";
  }
  function loadFeed() {
    fetch("/portal/feed")
      .then(function (r) { return r.json(); })
      .then(renderFeed)
      .catch(function () {});
  }
  loadFeed();
  setInterval(loadFeed, 30000);

  /* ---------- module health (where learners stall) ---------- */
  function renderModuleHealth(d) {
    var host = $("mh-list");
    if (!host) return;
    if (d.error) { host.innerHTML = "<div class='err'>Could not load module health.</div>"; return; }
    if (d.status === "building") {
      host.textContent = "Analysing unit-level engagement — module " + d.progress.done + " of " + d.progress.total + "…";
      setTimeout(loadModuleHealth, 2500);
      return;
    }
    var reports = d.reports || [];
    if (!reports.length) {
      host.innerHTML = "<p class='muted'>Not enough unit-level activity yet — this fills in as learners work through modules.</p>";
      return;
    }
    var out = "";
    reports.forEach(function (r) {
      var band = r.retention >= 70 ? "good" : r.retention >= 40 ? "mid" : "bad";
      var max = Math.max.apply(null, r.funnel.concat([1]));
      /* mark the bar with the biggest onward loss in mango */
      var lossIdx = -1, worst = 0;
      for (var i = 0; i < r.funnel.length - 1; i++) {
        if (r.funnel[i] >= 3) {
          var loss = (r.funnel[i] - r.funnel[i + 1]) / r.funnel[i];
          if (loss > worst) { worst = loss; lossIdx = i; }
        }
      }
      var bars = r.funnel.map(function (v, j) {
        return "<i" + (j === lossIdx && r.stallUnit ? " class='low'" : "") +
          " style='height:" + Math.max(8, Math.round(v * 100 / max)) + "%'></i>";
      }).join("");
      out +=
        "<div class='mhrow'><span class='mhname'>" + esc(r.title) +
        "<em>" + r.starters + " started · " + r.finishers + " reached the end" +
        (r.stallUnit ? " · <b>" + r.stallLossPct + "% stop at “" + esc(r.stallUnit) + "”</b>" : "") +
        "</em></span>" +
        "<span class='funnel' title='Learners remaining at each unit'>" + bars + "</span>" +
        "<span class='mhret " + band + "'>" + r.retention + "% finish</span></div>";
    });
    host.innerHTML = out;
  }
  function loadModuleHealth() {
    fetch("/portal/module-health")
      .then(function (r) { return r.json(); })
      .then(renderModuleHealth)
      .catch(function () {});
  }
  loadModuleHealth();

  function loadReflections() {
    fetch("/portal/reflections")
      .then(function (r) { return r.json(); })
      .then(renderReflections)
      .catch(function () {
        $("reflect-state").innerHTML = "<div class='err'>Could not load reflections — try refreshing.</div>";
      });
  }
  loadReflections();
