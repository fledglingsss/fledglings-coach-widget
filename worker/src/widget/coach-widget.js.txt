  /* Safe mini-formatter: escapes everything via DOM text nodes, then
   * supports **bold** and line breaks only. No HTML injection path. */
  function renderInto(el, text) {
    var lines = String(text).split("\n");
    for (var i = 0; i < lines.length; i++) {
      if (i > 0) el.appendChild(document.createElement("br"));
      var parts = lines[i].split(/\*\*(.+?)\*\*/g);
      for (var p = 0; p < parts.length; p++) {
        if (!parts[p]) continue;
        if (p % 2 === 1) {
          var strong = document.createElement("strong");
          strong.textContent = parts[p];
          el.appendChild(strong);
        } else {
          el.appendChild(document.createTextNode(parts[p]));
        }
      }
    }
  }

  function addMsg(role, text, crisis) {
    var div = document.createElement("div");
    div.className = "fl-msg " + (role === "user" ? "fl-msg-user" : "fl-msg-coach") + (crisis ? " fl-msg-crisis" : "");
    renderInto(div, text);
    msgs.appendChild(div);
    msgs.scrollTop = msgs.scrollHeight;
    return div;
  }

  function sendText(text) {
    if (busy) return;
    text = String(text).replace(/\s+$/, "").slice(0, MAX_INPUT_CHARS);
    if (!text) return;
    removeChipRows();
    addMsg("user", text);
    history.push({ role: "user", content: text });
    saveHistory(history);
    setBusy(true);

    fetch(ENDPOINT + "/api/coach", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        learner_id: learnerId,
        session_id: sessionId,
        learner_name: firstName,
        page: (document.title || "").slice(0, 160),
        messages: history.slice(-MAX_TURNS_SENT),
      }),
    })
      .then(function (r) { return r.json(); })
      .then(function (data) {
        setBusy(false);
        var reply = data && typeof data.reply === "string" && data.reply
          ? data.reply
          : "Sorry — something went wrong. Try again in a minute.";
        addMsg("coach", reply, data && data.kind === "crisis");
        history.push({ role: "assistant", content: reply });
        saveHistory(history);
      })
      .catch(function () {
        setBusy(false);
        addMsg("coach", "I can't reach the coach right now — give it another go in a minute.\n" + SUPPORT_TEXT, true);
      });
  }

  /* Wizard flow: three chip questions, then the pathway API. The
   * conversation history gets a compact summary so the model coach has
   * context if the learner keeps chatting afterwards. */
  function startWizard() {
    removeChipRows();
    addMsg("user", "Find my pathway 🧭");
    var answers = {};
    askStep("stage", function () {
      askStep("area", function () {
        askFocus(function () {
          submitWizard(answers);
        });
      });
    });

    function askStep(key, next) {
      var step = WIZARD_STEPS[key];
      addMsg("coach", step.question);
      chipRow(step.options, function (opt) {
        removeChipRows();
        addMsg("user", opt.label);
        answers[key] = opt.value;
        next();
      });
    }
    function askFocus(next) {
      var step = WIZARD_STEPS.focus[answers.area];
      addMsg("coach", step.question);
      chipRow(step.options, function (opt) {
        removeChipRows();
        addMsg("user", opt.label);
        answers.focus = opt.value;
        next();
      });
    }
  }

  function submitWizard(answers) {

        learner_id: learnerId,
        session_id: sessionId,
        stage: answers.stage,
        area: answers.area,
        focus: answers.focus,
      }),

      .then(function (r) { return r.json(); })
      .then(function (data) {

        var recs = (data && data.recommendations) || [];
        if (!recs.length) {
          addMsg("coach", "Sorry — I couldn't build your pathway just now. Try again in a minute.");
          return;
        }
        var lines = ["Here's your pathway — in this order:"];
        for (var i = 0; i < recs.length; i++) {
          lines.push((i + 1) + ". **" + recs[i].title + "** — " + recs[i].why);
        }
        var summary = lines.join("\n");
        addMsg("coach", summary);
        history.push({ role: "user", content: "I used the pathway finder. Stage: " + answers.stage + ", area: " + answers.area + ", focus: " + answers.focus + "." });
        history.push({ role: "assistant", content: summary });
        saveHistory(history);

        /* Explicit-consent step: nothing is ever added to a learner's
         * account unless they tap the specific module by name. */
        if (data.can_enrol && learnerEmail) {
          offerEnrolment(recs);
        } else {
          addMsg("coach", "You can start any of these from the course catalogue — just search the name.");
        }
      })
      .catch(function () {
        setBusy(false);
        addMsg("coach", "I couldn't reach the pathway service — give it another go in a minute.");
      });
  }

  function offerEnrolment(recs) {
    addMsg("coach", "Would you like me to add any of these to your dashboard? Nothing gets added unless you choose it.");
    chipRow(
      [{ label: "Yes — let me choose ✅", value: "yes" }, { label: "No thanks", value: "no" }],
      function (opt) {
        removeChipRows();
        addMsg("user", opt.label);
        if (opt.value === "no") {
          addMsg("coach", "No problem — they'll be in the course catalogue whenever you're ready.");
          return;
        }
        addMsg("coach", "Tap a module to add it — each one is added only when you tap it. Tap **Done** when you're finished.");
        showEnrolChips(recs.map(function (r) { return r.title; }));
      },
    );
  }

  function showEnrolChips(titles) {
    var options = titles.map(function (t) {
      return { label: "➕ " + t, value: t };
    });
    options.push({ label: "Done ✔", value: "__done__" });
    chipRow(options, function (opt) {
      removeChipRows();
      if (opt.value === "__done__") {
        addMsg("coach", "Sorted! Refresh your dashboard to see anything you added. Ask me if you want a hand getting started.");
        return;
      }
      addMsg("user", "Add " + opt.value);
      setBusy(true);
      fetch(ENDPOINT + "/api/enrol", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          learner_id: learnerId,
          session_id: sessionId,
          email: learnerEmail,
          title: opt.value,
        }),
      })
        .then(function (r) { return r.json(); })
        .then(function (data) {
          setBusy(false);
          var remaining = titles.filter(function (t) { return t !== opt.value; });
          if (data && data.ok) {
            addMsg("coach", "✅ **" + opt.value + "** is on your dashboard.");
          } else if (data && data.reason === "account_not_found") {
            addMsg("coach", "I couldn't match your account automatically — you can add it yourself from the course catalogue.");
          } else if (data && data.reason === "daily_cap") {
            addMsg("coach", "That's your adding done for today — the rest are in the course catalogue whenever you want them.");
            return;
          } else {
            addMsg("coach", "That one didn't go through — find **" + opt.value + "** in the course catalogue instead.");
            remaining = titles.filter(function (t) { return t !== opt.value; });
          }
          if (remaining.length) {
            showEnrolChips(remaining);
          } else {
            addMsg("coach", "That's all of them! Refresh your dashboard and they'll be waiting.");
          }
        })
        .catch(function () {
          setBusy(false);
          addMsg("coach", "I couldn't reach the enrolment service — you can add modules from the course catalogue instead.");
        });
    });
  }

  function renderHistory() {
    msgs.innerHTML = "";
    if (history.length === 0) {
      addMsg("coach",
        (firstName ? "Hi " + firstName + "! " : "Hi! ") +
        "I'm **Fledge**, your Fledglings coach. Money, work, confidence, staying safe online — ask me anything, or pick a quick start:");
      renderChips();
    } else {
      for (var i = 0; i < history.length; i++) {
        addMsg(history[i].role === "user" ? "user" : "coach", history[i].content);
      }
    }
  }

  /* Identity is a signed token, not the address itself: the widget
   * exchanges the LearnWorlds email for one (this page is served by
   * the school, so the mint is allowed) and reuses it afterwards.
   * Note this store belongs to the SCHOOL's origin — the tools live on
   * the worker's origin and mint their own; the two are independent by
   * design, and both are bound to this browser. */
  /* A token past its expiry must be discarded, not returned for ever —
   * otherwise the nudge silently dies after 30 days and can never
   * re-mint because the stale value is always found first. */
  function tokenAlive(token) {
    try {
      var payload = String(token).split(".")[0];
      var json = decodeURIComponent(
        atob(payload.replace(/-/g, "+").replace(/_/g, "/"))
          .split("")
          .map(function (c) { return "%" + c.charCodeAt(0).toString(16).padStart(2, "0"); })
          .join(""),
      );
      var claims = JSON.parse(json);
      return !!claims && typeof claims.exp === "number" && claims.exp * 1000 > Date.now();
    } catch (e) {
      return false;
    }
  }

  function ensureToken() {
    var stored = "";
    try { stored = localStorage.getItem("fl_hub_token_v1") || ""; } catch (e) { /* private mode */ }
    if (stored && !tokenAlive(stored)) {
      try { localStorage.removeItem("fl_hub_token_v1"); } catch (e) { /* private mode */ }
      stored = "";
    }
    if (stored) return Promise.resolve(stored);
    if (!learnerEmail) return Promise.resolve("");
    return fetch(ENDPOINT + "/api/identity", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ learner_id: learnerId, email: learnerEmail }),
    })
      .then(function (r) { return r.json(); })
      .then(function (d) {
        if (!d || !d.ok || !d.token) return "";
        try { localStorage.setItem("fl_hub_token_v1", d.token); } catch (e) { /* private mode */ }
        return d.token;
      })
      .catch(function () { return ""; });
  }

  function offerNextStep() {
    if (!learnerEmail) return;
    try {
      if (sessionStorage.getItem("fl_coach_nextstep_v1")) return;
      sessionStorage.setItem("fl_coach_nextstep_v1", "1");
    } catch (e) { /* private mode: still offer, just maybe twice */ }
    ensureToken().then(function (token) {
      if (!token) return;
      return fetch(ENDPOINT + "/api/next-step", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ learner_id: learnerId, token: token }),
      })
        .then(function (r) { return r.json(); })
        .then(function (d) {
          if (!d || !d.ok || !d.title) return;
          var lead = d.percent > 0
            ? "Pick up where you left off — you're " + d.percent + "% through **" + d.title + "**:"
            : "Ready for your next step? **" + d.title + "** is waiting:";
          addLinkMsg(lead, "Continue " + (d.percent > 0 ? "→" : "▸"), d.url);
        });
    }).catch(function () { /* a missing nudge is never an error */ });
  }

  function setOpen(next) {
    open = next;
    panel.className = open ? "fl-open" : "";
    btn.setAttribute("aria-expanded", String(open));
    if (open) { renderHistory(); offerNextStep(); input.focus(); }
    else { btn.focus(); }
  }
  btn.addEventListener("click", function () { setOpen(!open); });
  document.getElementById("fl-coach-close").addEventListener("click", function () { setOpen(false); });
  document.addEventListener("keydown", function (e) {
    if (e.key === "Escape" && open) setOpen(false);
  });

  var busy = false;
  function setBusy(next) {
    busy = next;
    send.disabled = next;
    var t = document.getElementById("fl-coach-typing");
    if (next && !t) {
      t = document.createElement("div");
      t.id = "fl-coach-typing";
      t.className = "fl-typing";
      t.setAttribute("aria-label", "Fledge is typing");
      t.innerHTML = "<span></span><span></span><span></span>";
      msgs.appendChild(t);
      msgs.scrollTop = msgs.scrollHeight;
    } else if (!next && t) {
      t.parentNode.removeChild(t);
    }
  }

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    var text = input.value.trim();
    if (!text) return;
    input.value = "";
    sendText(text);
  });

  input.addEventListener("keydown", function (e) {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      if (typeof form.requestSubmit === "function") form.requestSubmit();
      else form.dispatchEvent(new Event("submit", { cancelable: true }));
    }
  });

  fetch(ENDPOINT + "/health")

      if (!h || h.ok !== true) root.parentNode.removeChild(root);
    })
    .catch(function () {
      if (root.parentNode) root.parentNode.removeChild(root);
    });
